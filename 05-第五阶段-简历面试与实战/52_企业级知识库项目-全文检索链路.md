# 52. 企业级知识库项目：全文检索链路

> 阶段：第五阶段 | 前置知识：ElasticSearch、混合检索 RAG

## 为什么需要全文检索？

纯向量检索（RAG）擅长语义匹配，但有短板：
- 精确关键词搜索差（如产品型号、人名、错误码）
- 无法做复杂条件过滤（时间范围、文档类型、作者）
- 同义词/缩写可能漏召回

**全文检索（ElasticSearch）** 补全这些能力，和向量检索结合形成**混合检索**。

## 检索链路架构

```
用户查询
    ↓
Query 预处理（分词、纠错、扩展）
    ↓
┌─────────────┐    ┌─────────────┐
│  向量检索     │    │  全文检索     │
│  (Milvus)    │    │  (ES)        │
└──────┬──────┘    └──────┬──────┘
       │                    │
       └────────┬───────────┘
                ↓
          合并去重（Reciprocal Rank Fusion）
                ↓
          Reranker 重排（BGE-Reranker）
                ↓
          Top-K 结果 → 拼接 Prompt → 大模型生成
```

### 1. ES 索引设计

```python
# es_index.py
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

INDEX_NAME = "knowledge_base"

# 创建索引（中文 IK 分词）
index_mapping = {
    "mappings": {
        "properties": {
            "doc_id": {"type": "keyword"},
            "chunk_id": {"type": "keyword"},
            "content": {
                "type": "text",
                "analyzer": "ik_max_word",      # 索引时用 ik_max_word（细粒度）
                "search_analyzer": "ik_smart"    # 搜索时用 ik_smart（粗粒度）
            },
            "title": {"type": "text", "analyzer": "ik_max_word"},
            "doc_type": {"type": "keyword"},
            "category": {"type": "keyword"},
            "author": {"type": "keyword"},
            "page": {"type": "integer"},
            "created_at": {"type": "date"},
            "tags": {"type": "keyword"}
        }
    }
}

es.indices.create(index=INDEX_NAME, body=index_mapping, ignore=400)
print("索引创建成功")
```

### 2. 文档索引写入

```python
# es_indexer.py
from elasticsearch import Elasticsearch
from elasticsearch.helpers import bulk

es = Elasticsearch("http://localhost:9200")

def index_chunks(doc_id, chunks, metadata):
    """批量索引文档分块"""
    actions = []
    for i, chunk in enumerate(chunks):
        actions.append({
            "_index": "knowledge_base",
            "_id": f"{doc_id}_{i}",
            "_source": {
                "doc_id": doc_id,
                "chunk_id": f"{doc_id}_{i}",
                "content": chunk.page_content,
                "title": metadata.get("title", ""),
                "doc_type": metadata.get("file_type", ""),
                "category": metadata.get("category", ""),
                "author": metadata.get("author", ""),
                "page": chunk.metadata.get("page", 0),
                "tags": metadata.get("tags", []),
                "created_at": metadata.get("created_at", "")
            }
        })

    success, failed = bulk(es, actions)
    print(f"索引完成: 成功 {success}, 失败 {len(failed)}")
    return success
```

### 3. 全文检索

```python
# es_search.py
from elasticsearch import Elasticsearch

es = Elasticsearch("http://localhost:9200")

def keyword_search(query, filters=None, top_k=10):
    """
    全文检索 + 条件过滤

    Args:
        query: 查询文本
        filters: 过滤条件，如 {"doc_type": "pdf", "category": "技术文档"}
        top_k: 返回数量
    """
    # 多字段匹配：title 权重 3 倍，content 权重 1 倍
    must_clauses = [
        {
            "multi_match": {
                "query": query,
                "fields": ["title^3", "content"],
                "type": "best_fields"
            }
        }
    ]

    # 过滤条件
    filter_clauses = []
    if filters:
        for key, value in filters.items():
            filter_clauses.append({"term": {key: value}})

    search_body = {
        "query": {
            "bool": {
                "must": must_clauses,
                "filter": filter_clauses
            }
        },
        "highlight": {
            "fields": {
                "content": {"pre_tags": ["*"], "post_tags": ["*"]},
                "title": {"pre_tags": ["*"], "post_tags": ["*"]}
            }
        },
        "size": top_k
    }

    result = es.search(index="knowledge_base", body=search_body)

    chunks = []
    for hit in result["hits"]["hits"]:
        source = hit["_source"]
        highlight = hit.get("highlight", {})
        chunks.append({
            "chunk_id": source["chunk_id"],
            "doc_id": source["doc_id"],
            "content": source["content"],
            "title": source["title"],
            "score": hit["_score"],
            "highlight": highlight.get("content", [""])[0],
            "source": "keyword"
        })

    return chunks
```

### 4. 混合检索（向量 + 全文）

```python
# hybrid_search.py
from es_search import keyword_search
from langchain_community.vectorstores import PGVector
from langchain_openai import OpenAIEmbeddings

embeddings = OpenAIEmbeddings()
vectorstore = PGVector(
    collection_name="knowledge_base",
    connection_string="postgresql+psycopg2://user:password@localhost:5432/kb",
    embedding_function=embeddings
)

def vector_search(query, top_k=10):
    """向量检索"""
    results = vectorstore.similarity_search_with_score(query, k=top_k)
    chunks = []
    for doc, score in results:
        chunks.append({
            "chunk_id": doc.metadata.get("chunk_id", ""),
            "content": doc.page_content,
            "score": float(score),
            "source": "vector"
        })
    return chunks

def reciprocal_rank_fusion(results_list, k=60):
    """
    RRF 融合算法：把多个检索结果的排名合并
    score = Σ 1 / (k + rank_i)
    """
    fused_scores = {}
    for results in results_list:
        for rank, item in enumerate(results):
            cid = item["chunk_id"]
            if cid not in fused_scores:
                fused_scores[cid] = {"item": item, "score": 0}
            fused_scores[cid]["score"] += 1 / (k + rank + 1)

    # 按融合分数排序
    sorted_results = sorted(fused_scores.values(), key=lambda x: x["score"], reverse=True)
    return [r["item"] for r in sorted_results]

def hybrid_search(query, filters=None, top_k=5):
    """混合检索：向量 + 全文，RRF 融合，Reranker 重排"""
    # 1. 多路召回
    vec_results = vector_search(query, top_k=20)
    kw_results = keyword_search(query, filters=filters, top_k=20)

    # 2. RRF 融合
    merged = reciprocal_rank_fusion([vec_results, kw_results])

    # 3. Reranker 重排（可选，提升准确率）
    # from sentence_transformers import CrossEncoder
    # reranker = CrossEncoder('BAAI/bge-reranker-base')
    # pairs = [[query, item['content']] for item in merged[:20]]
    # scores = reranker.predict(pairs)
    # for item, score in zip(merged, scores):
    #     item['rerank_score'] = float(score)
    # merged.sort(key=lambda x: x.get('rerank_score', 0), reverse=True)

    return merged[:top_k]
```

### 5. RAG 问答

```python
# rag_qa.py
from hybrid_search import hybrid_search
from langchain_openai import ChatOpenAI

llm = ChatOpenAI(model="gpt-4o", temperature=0)

RAG_PROMPT = """你是一个专业的知识库问答助手。根据以下参考资料回答用户问题。

参考资料：
{context}

用户问题：{question}

回答要求：
1. 只根据参考资料回答，不要编造
2. 引用来源编号，如 [1] [2]
3. 如果参考资料中没有相关信息，回答"知识库中未找到相关信息"
4. 用中文回答，简洁准确
"""

def ask_question(question, filters=None):
    # 1. 混合检索
    chunks = hybrid_search(question, filters=filters, top_k=5)

    # 2. 拼接上下文
    context = ""
    for i, chunk in enumerate(chunks, 1):
        context += f"[{i}] (文档: {chunk.get('title', '')}) {chunk['content']}\n\n"

    # 3. 大模型生成
    prompt = RAG_PROMPT.format(context=context, question=question)
    response = llm.invoke(prompt)

    return {
        "answer": response.content,
        "sources": [{"title": c.get('title', ''), "score": c.get('score', 0)} for c in chunks]
    }
```

### 2. 高亮显示
ES 的 highlight 功能返回匹配关键词的高亮片段，前端可以展示命中位置。

### 3. 过滤条件
支持按文档类型、分类、作者、时间范围过滤，这是纯向量检索做不到的。

### 4. 同义词词典
IK 分词器支持自定义同义词词典，把"AI"和"人工智能"视为同义词，提升召回率。

## 学习要点

1. 企业级知识库必须用混合检索：向量（语义）+ 全文（关键词），互补短板
2. ES 索引用 ik_max_word 索引、ik_smart 搜索，中文效果好
3. RRF（Reciprocal Rank Fusion）是简单有效的结果融合算法
4. Reranker 重排是准确率提升的关键，推荐 BGE-Reranker
5. 全文检索支持复杂过滤（类型、分类、时间），这是向量检索没有的
6. Query 预处理（纠错、扩展、改写）能显著提升检索效果
